CREATE TABLE insumos(
id_insumos INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
nome VARCHAR(254),
unidade integer,
custo_unitario decimal(10,2),
estoque integer,
id_usuario integer,

 CONSTRAINT FK_INSUMOS
 FOREIGN KEY (ID_USUARIO) REFERENCES usuario(id_usuario)
)

CREATE TABLE insumos_utilizado(
id_insumos_utilizado INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
quantidade integer,
unidade integer,
id_insumos integer,
id_produto integer,

 CONSTRAINT FK_INSUMOS
 FOREIGN KEY (id_insumos) REFERENCES insumos(id_insumos)
 
 CONSTRAINT FK_PRODUTO
 FOREIGN KEY (id_produto) REFERENCES produtos(id_produtos)
)

CREATE TABLE produtos (
    id_produto INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(254),
    margem_lucro NUMERIC(4,3),
    sugestao NUMERIC(4,3),
    preco NUMERIC(4,3),
    lucro_unidade NUMERIC(4,3),
    custo_total NUMERIC(4,3),
    id_usuario INTEGER,
    id_insumos_utilizados INTEGER,

    CONSTRAINT FK_USUARIO
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),

    CONSTRAINT FK_INSUMOS_UTILIZADOS
    FOREIGN KEY (id_insumos_utilizados) REFERENCES insumos_utilizado(id_insumos_utilizado)
);
